mysql> CREATE PROCEDURE sp_sell_car (
    ->     IN p_car_id INT,
    ->     IN p_customer_id INT,
    ->     IN p_employee_id INT,
    ->     IN p_sale_date DATE,
    ->     IN p_sale_price DECIMAL(12,2),
    ->     IN p_payment_method VARCHAR(30)
    -> )
    -> BEGIN
    ->     DECLARE v_sale_id INT;
    ->     DECLARE v_car_price DECIMAL(12,2);
    ->     DECLARE v_status VARCHAR(20);
    ->
    ->     DECLARE EXIT HANDLER FOR SQLEXCEPTION
    ->     BEGIN
    ->         ROLLBACK;
    ->         RESIGNAL;
    ->     END;
    ->
    ->     START TRANSACTION;
    ->
    ->     SELECT car_status, price INTO v_status, v_car_price
    ->     FROM cars WHERE car_id = p_car_id FOR UPDATE;
    ->
    ->     IF v_status IS NULL OR v_status <> 'Available' THEN
    ->         SIGNAL SQLSTATE '45000'
    ->         SET MESSAGE_TEXT = 'Selected car is not available';
    ->     END IF;
    ->
    ->     IF p_sale_price <= 0 THEN
    ->         SIGNAL SQLSTATE '45000'
    ->         SET MESSAGE_TEXT = 'Sale price must be greater than zero';
    ->     END IF;
    ->
    ->     INSERT INTO sales (car_id,customer_id,employee_id,sale_date,sale_price)
    ->     VALUES (p_car_id,p_customer_id,p_employee_id,p_sale_date,p_sale_price);
    ->
    ->     SET v_sale_id = LAST_INSERT_ID();
    ->
    ->     INSERT INTO payments (sale_id,payment_date,amount,payment_method,payment_status)
    ->     VALUES (v_sale_id,p_sale_date,p_sale_price,p_payment_method,'Paid');
    ->
    ->     COMMIT;
    -> END//
